# ============================================================
#  Photobox – Einstellungen
#  Diese Datei liegt auf der SD-Karte (Laufwerk "bootfs") und
#  kann am PC mit einem Texteditor geändert werden.
#  Änderungen gelten nach dem nächsten Neustart des Pi.
# ============================================================

# Name des WLANs, das die Photobox aufspannt
WLAN_NAME=Photobox

# WLAN-Passwort (mindestens 8 Zeichen)
WLAN_PASSWORT=photobox123

# Normales WLAN (z. B. zu Hause), in das die Photobox auf der Statusseite
# umgeschaltet werden kann. Leer lassen, wenn nicht benötigt.
HEIM_WLAN_NAME=
HEIM_WLAN_PASSWORT=

# Womit die Photobox nach dem Einschalten startet:
#   hotspot = eigenes WLAN (Standard)
#   wlan    = normales WLAN; ist es nicht erreichbar, startet der Hotspot
START_MODUS=hotspot

# WLAN-Kanal (1-13) und Land
WLAN_KANAL=6
WLAN_LAND=DE

# CUPS-Druckername (leer = Standarddrucker)
DRUCKER=

# Druckoptionen, z. B. für Canon Selphy 10x15:  fit-to-page media=Postcard
DRUCK_OPTIONEN=fit-to-page

# Format des Fotostreifens: 58mm / 80mm (Bondrucker-Rolle) oder 10x15 (Fotopapier)
DRUCK_FORMAT=58mm

# Bondrucker mit ESC/POS (z. B. Citizen CT-S310II) direkt ansteuern: ja / nein
# ja = feinere Rasterung der Fotos und sauberer Schnitt am Ende des Streifens
BONDRUCKER_DIREKT=nein
